-- =====================================================
-- MotorWise - Car Dealership Management System
-- =====================================================

DROP DATABASE IF EXISTS motorwise;
CREATE DATABASE motorwise; 
USE motorwise;

-- -----------------------------------------------------
-- 1. ROLES (supports role-based access control)
-- -----------------------------------------------------
CREATE TABLE roles (
    role_id     TINYINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    role_name   VARCHAR(30) NOT NULL UNIQUE,   -- admin, staff, customer
    description VARCHAR(150)
);

-- -----------------------------------------------------
-- 2. USERS (login accounts for staff and customers)
-- -----------------------------------------------------
CREATE TABLE users (
    user_id       INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    role_id       TINYINT UNSIGNED NOT NULL,
    email         VARCHAR(120) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,       -- store a bcrypt/argon2 hash, NEVER plain text
    is_active     BOOLEAN NOT NULL DEFAULT TRUE,
    created_at    TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    last_login    TIMESTAMP NULL,
    CONSTRAINT fk_users_role FOREIGN KEY (role_id) REFERENCES roles(role_id)
);

-- -----------------------------------------------------
-- 3. CUSTOMERS (profile info; linked to a login account)
-- -----------------------------------------------------
CREATE TABLE customers (
    customer_id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    user_id     INT UNSIGNED UNIQUE,           -- NULL allowed for walk-in customers
    first_name  VARCHAR(50) NOT NULL,
    last_name   VARCHAR(50) NOT NULL,
    phone       VARCHAR(20),
    address     VARCHAR(150),
    city        VARCHAR(60),
    state       CHAR(2),
    zip_code    VARCHAR(10),
    created_at  TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_customers_user FOREIGN KEY (user_id) REFERENCES users(user_id)
        ON DELETE SET NULL
);

-- -----------------------------------------------------
-- 4. VEHICLES (inventory)
-- -----------------------------------------------------
CREATE TABLE vehicles (
    vehicle_id    INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    vin           CHAR(17) NOT NULL UNIQUE,
    make          VARCHAR(40) NOT NULL,
    model         VARCHAR(40) NOT NULL,
    model_year    SMALLINT UNSIGNED NOT NULL,
    body_type     VARCHAR(30),                 -- sedan, SUV, truck...
    color         VARCHAR(30),
    mileage       INT UNSIGNED NOT NULL DEFAULT 0,
    fuel_type     VARCHAR(20),                 -- gas, hybrid, electric...
    transmission  VARCHAR(20),
    price         DECIMAL(10,2) NOT NULL CHECK (price >= 0),
    status        ENUM('available','pending','sold','removed') NOT NULL DEFAULT 'available',
    description   TEXT,
    image_url     VARCHAR(255),
    created_by    INT UNSIGNED,                -- staff user who added it
    created_at    TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at    TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    CONSTRAINT fk_vehicles_creator FOREIGN KEY (created_by) REFERENCES users(user_id)
        ON DELETE SET NULL,
    INDEX idx_vehicle_search (make, model, price),   -- speeds up search and filtering
    INDEX idx_vehicle_status (status)
);

-- -----------------------------------------------------
-- 5. FEATURES (for "search by features" and comparison)
-- -----------------------------------------------------
CREATE TABLE features (
    feature_id   INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    feature_name VARCHAR(60) NOT NULL UNIQUE   -- Sunroof, Backup Camera, AWD...
) ENGINE=InnoDB;

CREATE TABLE vehicle_features (
    vehicle_id INT UNSIGNED NOT NULL,
    feature_id INT UNSIGNED NOT NULL,
    PRIMARY KEY (vehicle_id, feature_id),
    CONSTRAINT fk_vf_vehicle FOREIGN KEY (vehicle_id) REFERENCES vehicles(vehicle_id) ON DELETE CASCADE,
    CONSTRAINT fk_vf_feature FOREIGN KEY (feature_id) REFERENCES features(feature_id) ON DELETE CASCADE
);

-- -----------------------------------------------------
-- 6. TRANSACTIONS (purchases + receipt data)
--    No card numbers are stored (PCI DSS). Only the
--    reference ID returned by the payment sandbox.
-- -----------------------------------------------------
CREATE TABLE transactions (
    transaction_id    INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    receipt_number    VARCHAR(20) NOT NULL UNIQUE,   -- e.g. MW-20261105-0001
    customer_id       INT UNSIGNED NOT NULL,
    vehicle_id        INT UNSIGNED NOT NULL,
    processed_by      INT UNSIGNED,                  -- staff user, NULL if online
    sale_price        DECIMAL(10,2) NOT NULL,
    tax_amount        DECIMAL(10,2) NOT NULL DEFAULT 0.00,
    total_amount      DECIMAL(10,2) NOT NULL,
    payment_method    VARCHAR(30) NOT NULL DEFAULT 'sandbox_card',
    payment_reference VARCHAR(100),                  -- sandbox confirmation/token ID
    payment_status    ENUM('pending','completed','failed','refunded') NOT NULL DEFAULT 'pending',
    transaction_date  TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_tx_customer FOREIGN KEY (customer_id) REFERENCES customers(customer_id),
    CONSTRAINT fk_tx_vehicle  FOREIGN KEY (vehicle_id)  REFERENCES vehicles(vehicle_id),
    CONSTRAINT fk_tx_staff    FOREIGN KEY (processed_by) REFERENCES users(user_id) ON DELETE SET NULL,
    INDEX idx_tx_customer (customer_id),
    INDEX idx_tx_vehicle (vehicle_id)
);
